# Python常用代码示例合集

## 实例方法，类方法，静态方法比较
```python
class Person:
    species = "人类"  # 类属性（所有实例共享）

    def __init__(self, name):
        self.name = name  # 实例属性

    # 1. 实例方法：需要操作具体对象的数据
    def say_hello(self):
        print(f"你好，我是 {self.name}")

    # 2. 类方法：需要操作类本身的数据或构造特殊实例
    @classmethod
    def get_species(cls):
        print(f"我是 {cls.species}")

    @classmethod
    def create_anonymous(cls):
        return cls("匿名用户")  # 用类方法作为“工厂”，创建实例

    # 3. 静态方法：不需要任何类或实例数据，只是逻辑上属于该类
    @staticmethod
    def is_adult(age):
        return age >= 18

# 调用演示
p1 = Person("张三")

p1.say_hello()                # 输出：你好，我是张三（实例方法）
Person.get_species()          # 输出：我是 人类（类方法，直接通过类调用）
p1.get_species()              # 也可以通过实例调用，但不推荐

anon = Person.create_anonymous() # 类方法作为工厂
anon.say_hello()              # 输出：你好，我是 匿名用户

print(Person.is_adult(20))    # 输出：True（静态方法，像普通函数一样用）
print(p1.is_adult(16))        # 输出：False（也可以通过实例调用）
```

## 单例模式设计方法

### 重写__new__()方法

```python
class Single1(object):  
    # 通过重写__new__()实现单例模式  
    obj = None  
    def __new__(cls, *args, **kwargs):  
        if cls.obj is None:  
            cls.obj = super(Single1, cls).__new__(cls)  
        return cls.obj
        
if __name__ == '__main__':  
    # 第一种单例模式写法  
    s1 = Single1()  
    print(f"s1内存地址：{s1}")  
    s2 = Single1()  
    print(f"s2内存地址：{s2}")
```


## 互斥锁

```python
from threading import Thread,Lock  
  
a = 0  
b = 10000000  
  
lock = Lock()  
def add1():  
    lock.acquire()  
    for i in range(b):  
        global a  
        a += 1  
    print(a)  
    lock.release()  
  
  
def add2():  
    lock.acquire()  
    for i in range(b):  
        global a  
        a += 1  
    print(a)  
    lock.release()  
  
if __name__ == '__main__':  
    first = Thread(target=add1)  
    second = Thread(target=add2)  
    first.start()  
    second.start()
```

## 进程Process用法

```python
from multiprocessing import Process, Manager  
import os  
  
def sing(name):  
    print(f'{name} sing')  
    print(f"Process ID of child process 1：{os.getpid()}")  
    print(f"Parent process ID：{os.getppid()}")  
def dance(name):  
    print(f'{name} dance')  
    print(f"Process ID of child process 2：{os.getpid()}")  
    print(f"Parent process ID：{os.getppid()}")  
  
if __name__ == '__main__':  
    p1 = Process(target=sing,name='child process 1', args=("Bob",))  
    p2 = Process(target=dance, name='child process 2', args=("John",))  
    p1.start()  
    p2.start()  
    print(f"Process Name：{p1.name}")
```

## 进程间的通信

```python
from multiprocessing import Queue, Process  
  
  
def read(que):  
    while True:  
        item = que.get()  
        if item is None:   # 哨兵  
            break  
        print(f"队列消息：{item}")  
  
def write(que):  
    for i in range(10):  
        que.put(i)  
    que.put(None)  # 发送结束信号  
  
if __name__ == '__main__':  
    que = Queue(3)  
    p1 = Process(target=read, args=(que,))  
    p2 = Process(target=write, args=(que,))  
    p2.start()  
    p1.start()
"""
输出：
队列消息：0
队列消息：1
队列消息：2
队列消息：3
队列消息：4
队列消息：5
队列消息：6
队列消息：7
队列消息：8
队列消息：9
"""

"""
需要注意的是，在代码中，我加入了哨兵机制，如果不加哨兵，在read进程读取完最后一个元素后，生产者已经完成了生产，此时不会再往队列中放入新消息，但是消费者依旧在进行get操作，由于没有新元素进行get，导致read进程阻塞
```

```python
from gevent import monkey
monkey.patch_all()  # 在所有其他导入前执行[reference:29]

import gevent
import urllib.request

def fetch_url(url):
    print(f'开始请求: {url}')
    resp = urllib.request.urlopen(url)
    data = resp.read()
```
